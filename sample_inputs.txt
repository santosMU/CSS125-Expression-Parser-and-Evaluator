# exprscope demonstration inputs

# 1. operator precedence
2 + 3 * 4

# 2. parentheses
(2 + 3) * 4

# 3. right-associative exponentiation
2 ^ 3 ^ 2

# 4. lexical binding
let x = 5 in x * 3

# 5. scope and shadowing
let x = 10 in let x = 2 in x + 1

# 6. conditional expression
if 3 < 5 then 100 else 0

# 7. boolean expression
true and not false

# 8. recursive functions
fact(5) + fib(6)

# 9. built-in functions
max(4, 9) + sqrt(16)

# error cases

# type error
1 + true

# undefined identifier
x + 1

# syntax error
2 + * 3

# lexical error
2 $ 3

# runtime error
10 / 0

# function arity error
max(1)

# scope restoration and outer initializer binding: 21
let x = 10 in (let x = x + 1 in x) + x

# short circuit: false
false and (10 / 0 > 1)

# only the selected branch executes: 120
if 3 < 5 then fact(5) else 10 / 0

# runtime error that must not expose a python traceback
0 ^ -1
